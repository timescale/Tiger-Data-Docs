import * as C from "@constants";

import { Callout } from "@stainless-api/docs/components";


import SelfHostedRedhatXPlatform from "./_install-self-hosted-redhat-x-platform.mdx";

<NumberedList>

<NumberedItem title={`Install the latest ${C.PG} packages`}>

```bash
 sudo yum install https://download.postgresql.org/pub/repos/yum/reporpms/EL-$(rpm -E %{rhel})-x86_64/pgdg-redhat-repo-latest.noarch.rpm
 ```

</NumberedItem>

<NumberedItem title={`Add the ${C.TIMESCALE_DB} repository`}>

```bash
sudo tee /etc/yum.repos.d/timescale_timescaledb.repo <<EOL
[timescale_timescaledb]
name=timescale_timescaledb
baseurl=https://packagecloud.io/timescale/timescaledb/el/$(rpm -E %{rhel})/\$basearch
repo_gpgcheck=1
gpgcheck=0
enabled=1
gpgkey=https://packagecloud.io/timescale/timescaledb/gpgkey
sslverify=1
sslcacert=/etc/pki/tls/certs/ca-bundle.crt
metadata_expire=300
EOL
```

</NumberedItem>

<NumberedItem title="Update your local repository list">

```bash
sudo dnf makecache
```

</NumberedItem>

<NumberedItem title={`Install ${C.TIMESCALE_DB}`}>

To avoid errors, **do not** install {C.TDB_APACHE} and {C.TDB_COMMUNITY} at the same time.

```bash
sudo yum install timescaledb-2-postgresql-18 postgresql18
```

<Callout variant="note">

On Red Hat Enterprise Linux 8 and 9, disable the built-in {C.PG} module:

`sudo dnf -qy module disable postgresql`

</Callout>

</NumberedItem>
