import * as C from "@constants";
import { Steps, Step } from "@stainless-api/docs/components";

### Restoring data into a {C.SERVICE_LONG} with COPY

<Steps>

<Step title={`Connect to your ${C.SERVICE_LONG}:`}>

```sql
psql "postgres://tsdbadmin:<PASSWORD>@<HOST>:<PORT>/tsdb?sslmode=require"
```

</Step>

<Step title={`Restore the data to your ${C.SERVICE_LONG}:`}>

```sql
\copy <TABLE_NAME> FROM '<TABLE_NAME>.csv' WITH (FORMAT CSV);
```

Repeat for each table and hypertable you want to migrate.

</Step>

</Steps>
