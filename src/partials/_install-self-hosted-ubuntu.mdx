import * as C from "../constants";
import { Steps, Step } from "@stainless-api/docs/components";

import SelfHostedDebianStart from "./_install-self-hosted-debian-based-start.mdx";
import SelfHostedDebianEnd from "./_install-self-hosted-debian-based-end.mdx";

<SelfHostedDebianStart />

<Steps>

<Step title={`Add the ${C.TIMESCALE_DB} package`}>

```bash
echo "deb https://packagecloud.io/timescale/timescaledb/ubuntu/ $(lsb_release -c -s) main" | sudo tee /etc/apt/sources.list.d/timescaledb.list
```

</Step>

<Step title={`Install the ${C.TIMESCALE_DB} GPG key`}>

```bash
wget --quiet -O - https://packagecloud.io/timescale/timescaledb/gpgkey | sudo gpg --dearmor -o /etc/apt/trusted.gpg.d/timescaledb.gpg
```

For Ubuntu 21.10 and earlier use the following command:
```bash
wget --quiet -O - https://packagecloud.io/timescale/timescaledb/gpgkey | sudo apt-key add -
```

<SelfHostedDebianEnd />

</Step>

</Steps>
