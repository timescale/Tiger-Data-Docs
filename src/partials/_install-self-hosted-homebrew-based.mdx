import * as C from "@constants";
import { NumberedList, NumberedItem } from "@components/NumberedList";

<NumberedList>

<NumberedItem title="Install Homebrew">

If homebrew is not already installed on your machine, run the following command:
```bash
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
```

For more information about Homebrew, including installation instructions,
see the [Homebrew documentation](https://docs.brew.sh/Installation).

</NumberedItem>

<NumberedItem title={`Install and start ${C.PG}`}>

```bash
brew install postgresql@17
brew cleanup --prune-prefix
brew services start postgresql@17
```

</NumberedItem>

<NumberedItem title={`Add ${C.PG} to your path`}>

```bash
 echo 'export PATH="/opt/homebrew/opt/postgresql@17/bin:$PATH"' >> ~/.zshrc
 source ~/.zshrc
 ```

</NumberedItem>

<NumberedItem title={`Add the ${C.TIMESCALE_DB} Homebrew tap`}>

```bash
brew tap timescale/tap
```

</NumberedItem>

<NumberedItem title={`Install ${C.TIMESCALE_DB}`}>

```bash
brew install timescaledb
```

</NumberedItem>

<NumberedItem title="Configure your database">

```bash
timescaledb-tune --quiet --yes
```

</NumberedItem>

<NumberedItem title="Run the setup script to complete installation">

```bash
$(brew --prefix)/Cellar/timescaledb/$(brew list --versions timescaledb | awk '{print $2}')/bin/timescaledb_move.sh
```

This command automatically locates the correct version and runs the setup script.

</NumberedItem>

<NumberedItem title={`Start the ${C.PG} service`}>

```bash
brew services restart postgresql@17
```

</NumberedItem>

<NumberedItem title={`Verify ${C.PG} is running`}>

```bash
brew services list | grep postgresql
```

You should see `postgresql@17` with status `started`.

</NumberedItem>

<NumberedItem title={`Connect to ${C.PG}`}>

```bash
psql postgres
```

You are now in the psql shell.

</NumberedItem>

<NumberedItem title="Set the password for your database user">

(optional but recommended)

 ```bash
 \password
 ```

 When you have set the password, type `\q` to exit psql.

</NumberedItem>

</NumberedList>
