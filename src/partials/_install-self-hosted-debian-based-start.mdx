import * as C from "@constants";
import { NumberedList, NumberedItem } from "@components/NumberedList";

<NumberedItem title="Update your local repository list">

```bash
sudo apt update
```

A fresh install ships with empty package lists, so the next step fails without this.

</NumberedItem>

<NumberedItem title={`Install the latest ${C.PG} packages`}>

```bash
sudo apt install gnupg postgresql-common apt-transport-https lsb-release wget
```

</NumberedItem>

<NumberedItem title={`Run the ${C.PG} package setup script`}>

```bash
sudo /usr/share/postgresql-common/pgdg/apt.postgresql.org.sh
```

</NumberedItem>
