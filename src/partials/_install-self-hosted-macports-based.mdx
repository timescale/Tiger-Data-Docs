import * as C from "../constants";
import { Callout } from "@stainless-api/docs/components";
import { Steps, Step } from "@stainless-api/docs/components";

<Steps>

<Step title="Install MacPorts by downloading and running the package installer.">

For more information about MacPorts, including installation instructions,
see the [MacPorts documentation][macports].

</Step>

<Step title={`Install ${C.TIMESCALE_DB} and psql:`}>

```bash
sudo port install timescaledb libpqxx
```

To view the files installed, run:

```bash
port contents timescaledb libpqxx
```

<Callout variant="tip">

MacPorts does not install the `timescaledb-tools` package or run the `timescaledb-tune`
script. For more information about tuning your database, see the [{C.TIMESCALE_DB} tuning tool][timescale-tuner].

</Callout>

</Step>

<Step title={`Log in to ${C.PG} as \`postgres\``}>

```bash
sudo -u postgres psql
```
You are in the psql shell.

</Step>

<Step title="Set the password for `postgres`">

```bash
 \password postgres
 ```

 When you have set the password, type `\q` to exit psql.

</Step>

</Steps>
