import * as C from "@constants";
import { Callout } from "@stainless-api/docs/components";
import { NumberedList, NumberedItem } from "@components/NumberedList";


<NumberedItem title="Update your local repository list">

```bash
sudo yum update
```

</NumberedItem>

<NumberedItem title={`Install ${C.TIMESCALE_DB}`}>

To avoid errors, **do not** install {C.TDB_APACHE} and {C.TDB_COMMUNITY} at the same time.

```bash
sudo yum install timescaledb-2-postgresql-18 postgresql18
```

<Callout variant="note">

On Red Hat Enterprise Linux 8 and later, disable the built-in {C.PG} module:

`sudo dnf -qy module disable postgresql`

</Callout>

</NumberedItem>

<NumberedItem title={`Initialize the ${C.PG} instance`}>

```bash
sudo /usr/pgsql-18/bin/postgresql-18-setup initdb
```

</NumberedItem>

<NumberedItem title={`Tune your ${C.PG} instance for ${C.TIMESCALE_DB}`}>

```bash
sudo timescaledb-tune --pg-config=/usr/pgsql-18/bin/pg_config
```

This script is included with the `timescaledb-tools` package when you install {C.TIMESCALE_DB}.
For more information, see [configuration](/reference/configuration/).

</NumberedItem>

<NumberedItem title={`Enable and start ${C.PG}`}>

```bash
sudo systemctl enable postgresql-18
sudo systemctl start postgresql-18
```

</NumberedItem>

<NumberedItem title={`Log in to ${C.PG} as \`postgres\``}>

```bash
sudo -u postgres psql
```

You are now in the psql shell.

</NumberedItem>

<NumberedItem title={`Set the password for \`postgres\``}>

```bash
\password postgres
```

When you have set the password, type `\q` to exit psql.

</NumberedItem>
